    22→async function callZaiHttp(messages: ChatMessage[]): Promise<CompletionResult> {
    23→  const apiKey = process.env.ZAI_API_KEY
    24→  if (!apiKey) throw new Error('ZAI_API_KEY تنظیم نشده است.')
    25→  const res = await fetch(`${ZAI_HTTP_BASE}/chat/completions`, {
    26→    method: 'POST',
    27→    headers: {
    28→      'Content-Type': 'application/json',
    29→      Authorization: `Bearer ${apiKey}`,
    30→    },
    31→    body: JSON.stringify({
    32→      model: ZAI_HTTP_MODEL,
    33→      messages,
    34→      temperature: 0.7,
    35→      /* ارائه‌های ۹ تا ۱۴ اسلایدی JSON بزرگ می‌سازند؛ بدون این سقف،
    36→       * خروجی وسط JSON قطع می‌شد و ساختار نامعتبر می‌شد (باگ «بیش از ۸ اسلاید کار نمی‌کند») */
    37→      max_tokens: 8192,
    38→    }),
    39→    signal: AbortSignal.timeout(180_000),
    40→  })
    41→  if (!res.ok) {
    42→    const detail = await res.text().catch(() => '')
    43→    throw new Error(
    44→      `خطای سرویس Z.ai (کد ${res.status}). کلید API و اعتبار حساب را بررسی کنید. ${detail.slice(0, 160)}`
    45→    )
    46→  }
    47→  const data = (await res.json()) as {
    48→    choices?: { message?: { content?: string } }[]
    49→  }
    50→  return { content: data.choices?.[0]?.message?.content ?? '' }
    51→}

    53→async function callZaiSdk(messages: ChatMessage[]): Promise<CompletionResult> {
    54→  const { default: ZAI } = await import('z-ai-web-dev-sdk')
    55→  const zai = await ZAI.create()
    56→  const completion = await zai.chat.completions.create({
    57→    messages: messages as unknown as Parameters<typeof zai.chat.completions.create>[0]['messages'],
    58→    thinking: { type: 'disabled' },
    59→  })
    60→  return { content: completion.choices[0]?.message?.content ?? '' }
    61→}

    63→async function chat(messages: ChatMessage[]): Promise<string> {
    64→  const result = aiMode() === 'http' ? await callZaiHttp(messages) : await callZaiSdk(messages)
    65→  const content = result.content
    66→  if (!content || !content.trim()) throw new Error('پاسخ خالی از مدل هوش مصنوعی.')
    67→  return content
    68→}

    94→export function topicSeed(topic: string): number {
    95→  let h = 0
    96→  const t = topic ?? ''
    97→  for (let i = 0; i < t.length; i++) h = (h * 31 + t.charCodeAt(i)) >>> 0
    98→  return h
    99→}

   101→export function planSlideTypes(
   102→  slideCount: number,
   103→  visualMode: 'offline' | 'image',
   104→  seed: number,
   105→  preferredLayouts?: string[],
   106→): string[] {
   107→  const n = Math.max(4, Math.min(14, Math.floor(slideCount) || 8))
   108→  const plan: string[] = ['cover']
   109→  const hasAgenda = n >= 5
   110→  if (hasAgenda) plan.push('agenda')
   111→  const mid = n - 2 - (hasAgenda ? 1 : 0) /* تعداد اسلایدهای میانی */
   112→
   113→  /* جداکننده بخش برای ارائه‌های بلند — ریتم بصری می‌سازد */
   114→  const sections = mid >= 9 ? 2 : mid >= 6 ? 1 : 0
   115→  /* اسلاید تصویر تمام‌عرض فقط در حالت تصویری */
   116→  const images = visualMode === 'image' ? (mid >= 8 ? 2 : mid >= 5 ? 1 : 0) : 0
   117→  const coreCount = Math.max(0, mid - sections - images)
   118→
   119→  /* استخر چرخشی انواع میانی — bullets ستون فقرات است، انواع دیگر بین آن‌ها */
   120→  const baseCore =
   121→    visualMode === 'image'
   122→      ? ['bullets', 'two-column', 'stats', 'comparison', 'bullets', 'timeline', 'stats', 'bullets', 'two-column', 'timeline', 'comparison', 'quote']
   123→      : ['bullets', 'stats', 'two-column', 'bullets', 'comparison', 'timeline', 'stats', 'quote', 'bullets', 'two-column', 'timeline', 'comparison']
   124→
   125→  /* اولویت کاربر (پلن حرفه‌ای): انواع انتخابی قلب چیدمان را می‌سازند.
   126→   * با ۲ نوع و بیشتر، بین آن‌ها می‌چرخیم؛ با ۱ نوع، با bullets تناوب می‌کنیم. */
   127→  const pref = (preferredLayouts ?? []).filter((t, i, arr) => arr.indexOf(t) === i)
   128→  const core =
   129→    pref.length >= 2
   130→      ? Array.from({ length: Math.max(coreCount, 2) }, (_, i) => pref[i % pref.length])
   131→      : pref.length === 1
   132→        ? Array.from({ length: Math.max(coreCount, 2) }, (_, i) =>
   133→            i % 2 === 0 ? pref[0] : pref[0] === 'bullets' ? 'two-column' : 'bullets'
   134→          )
   135→        : baseCore
   136→
   137→  const picked: string[] = []
   138→  let i = Math.abs(seed) % core.length
   139→  let guard = 0
   140→  while (picked.length < coreCount && guard < coreCount * 20 + 40) {
   141→    guard++
   142→    const t = core[i % core.length]
   143→    i++
   144→    if (picked.length && picked[picked.length - 1] === t) continue
   145→    picked.push(t)
   146→  }
   147→  while (picked.length < coreCount) picked.push('bullets') /*Safety valve */
   148→
   149→  /* درج جداکننده‌ها و اسلایدهای تصویری در موقعیت‌های نسبی —
   150→   * از انتها درج می‌کنیم تا ایندکس‌های قبلی جابه‌جا نشوند؛
   151→   * اگر همسایه هم‌نوع بود، یک جای جلوتر می‌نشینیم */
   152→  const slots: [number, string][] = []
   153→  if (sections >= 1) slots.push([Math.max(1, Math.floor(picked.length / 3)), 'section'])
   154→  if (sections >= 2) slots.push([Math.max(3, Math.floor((picked.length * 2) / 3)), 'section'])
   155→  if (images >= 1) slots.push([Math.max(0, Math.floor(picked.length * 0.55)), 'image'])
   156→  if (images >= 2) slots.push([Math.max(1, Math.floor(picked.length * 0.85)), 'image'])
   157→  slots.sort((a, b) => b[0] - a[0])
   158→  for (const [pos, type] of slots) {
   159→    let p = pos
   160→    while (p < picked.length && (picked[p] === type || (p > 0 && picked[p - 1] === type))) p++
   161→    picked.splice(Math.min(p, picked.length), 0, type)
   162→  }
   163→
   164→  plan.push(...picked)
   165→  plan.push('conclusion')
   166→  return plan
   167→}

   315→export function extractJSON(text: string): unknown {
   316→  let t = text.trim()
   317→  t = t.replace(/^```(?:json)?\s*/i, '').replace(/```\s*$/i, '')
   318→  // 1) whole text might be pure JSON
   319→  try {
   320→    return JSON.parse(t)
   321→  } catch {
   322→    /* fall through */
   323→  }
   324→  // 2) slice the outermost braces
   325→  const start = t.indexOf('{')
   326→  const end = t.lastIndexOf('}')
   327→  if (start === -1 || end === -1 || end <= start) throw new Error('پاسخ مدل JSON معتبر نبود.')
   328→  const slice = t.slice(start, end + 1)
   329→  try {
   330→    return JSON.parse(slice)
   331→  } catch {
   332→    // 3) last resort: progressively try balanced-object candidates
   333→    for (let i = start; i < end; i++) {
   334→      if (t[i] !== '{') continue
   335→      const cand = t.slice(i, end + 1)
   336→      try {
   337→        return JSON.parse(cand)
   338→      } catch {
   339→        /* keep looking */
   340→      }
   341→    }
   342→    throw new Error('پاسخ مدل JSON معتبر نبود.')
   343→  }
   344→}

   354→/* لینک تصویر امن — فقط http(s)، حداکثر ۵۰۰ کاراکتر، بدون نویسه خطرناک */
   355→function safeImageUrl(v: unknown): string | undefined {
   356→  const s = String(v ?? '').trim()
   357→  if (!s || s.length > 500) return undefined
   358→  if (!/^https?:\/\/[\w\-._~:\/?#[\]@!$&'()*+,;=%]+$/i.test(s)) return undefined
   359→  return s
   360→}

   362→function normSlide(raw: Record<string, unknown>): DeckSlide | null {
   363→  const type = String(raw.type ?? '')
   364→  switch (type) {
   365→    case 'cover':
   366→      return {
   367→        type: 'cover',
   368→        badge: cut(raw.badge, 40) || undefined,
   369→        title: cut(raw.title, 90),
   370→        subtitle: cut(raw.subtitle, 200) || undefined,
   371→        /* نام ارائه‌دهنده/اعضا دیگر پذیرفته نمی‌شود */
   372→        date: cut(raw.date, 40) || undefined,
   373→        image: safeImageUrl(raw.image),
   374→        imageCaption: cut(raw.imageCaption ?? raw.caption, 120) || undefined,
   375→      }
   376→    case 'image': {
   377→      const img = safeImageUrl(raw.image)
   378→      if (!img) return null // image slide without a valid image is useless
   379→      return {
   380→        type: 'image',
   381→        kicker: cut(raw.kicker, 40) || undefined,
   382→        title: cut(raw.title, 80),
   383→        image: img,
   384→        caption: cut(raw.caption, 140) || undefined,
   385→        text: cut(raw.text, 220) || undefined,
   386→      }
   387→    }
   388→    case 'agenda':
   389→      return {
   390→        type: 'agenda',
   391→        kicker: cut(raw.kicker, 40) || undefined,
   392→        title: cut(raw.title, 80),
   393→        items: Array.isArray(raw.items) ? raw.items.slice(0, 8).map((i) => cut(i, 90)).filter(Boolean) : [],
   394→      }
   395→    case 'bullets':
   396→      return {
   397→        type: 'bullets',
   398→        kicker: cut(raw.kicker, 40) || undefined,
   399→        title: cut(raw.title, 80),
   400→        items: Array.isArray(raw.items)
   401→          ? raw.items.slice(0, 6).map((it) => {
   402→              const o = (it ?? {}) as Record<string, unknown>
   403→              return {
   404→                title: cut(o.title, 60) || undefined,
   405→                text: cut(o.text, 240),
   406→                image: safeImageUrl(o.image),
   407→                icon: isDeckIcon(o.icon) ? o.icon : undefined,
   408→              }
   409→            }).filter((i) => i.text)
   410→          : [],
   411→      }
   412→    case 'two-column':
   413→      return {
   414→        type: 'two-column',
   415→        kicker: cut(raw.kicker, 40) || undefined,
   416→        title: cut(raw.title, 80),
   417→        columns: Array.isArray(raw.columns)
   418→          ? raw.columns.slice(0, 2).map((c) => {
   419→              const o = (c ?? {}) as Record<string, unknown>
   420→              return {
   421→                heading: cut(o.heading, 50),
   422→                items: Array.isArray(o.items) ? o.items.slice(0, 6).map((i) => cut(i, 110)).filter(Boolean) : [],
   423→              }
   424→            })
   425→          : [],
   426→      }
   427→    case 'stats':
   428→      return {
   429→        type: 'stats',
   430→        kicker: cut(raw.kicker, 40) || undefined,
   431→        title: cut(raw.title, 80),
   432→        stats: Array.isArray(raw.stats)
   433→          ? raw.stats.slice(0, 4).map((st) => {
   434→              const o = (st ?? {}) as Record<string, unknown>
   435→              return { value: cut(o.value, 14), label: cut(o.label, 60), hint: cut(o.hint, 90) || undefined }
   436→            }).filter((s) => s.value)
   437→          : [],
   438→        footnote: cut(raw.footnote, 170) || undefined,
   439→      }
   440→    case 'quote':
   441→      return {
   442→        type: 'quote',
   443→        quote: cut(raw.quote, 240),
   444→        author: cut(raw.author, 60) || undefined,
   445→        role: cut(raw.role, 70) || undefined,
   446→      }
   447→    case 'timeline':
   448→      return {
   449→        type: 'timeline',
   450→        kicker: cut(raw.kicker, 40) || undefined,
   451→        title: cut(raw.title, 80),
   452→        steps: Array.isArray(raw.steps)
   453→          ? raw.steps.slice(0, 5).map((st) => {
   454→              const o = (st ?? {}) as Record<string, unknown>
   455→              return { label: cut(o.label, 6) || '•', title: cut(o.title, 60), text: cut(o.text, 130) }
   456→            }).filter((s) => s.title)
   457→          : [],
   458→      }
   459→    case 'conclusion':
   460→      return {
   461→        type: 'conclusion',
   462→        title: cut(raw.title, 80),
   463→        points: Array.isArray(raw.points) ? raw.points.slice(0, 5).map((p) => cut(p, 120)).filter(Boolean) : [],
   464→        cta: cut(raw.cta, 70) || undefined,
   465→      }
   466→    case 'section':
   467→      return {
   468→        type: 'section',
   469→        kicker: cut(raw.kicker, 40) || undefined,
   470→        title: cut(raw.title, 90),
   471→        subtitle: cut(raw.subtitle, 160) || undefined,
   472→      }
   473→    case 'comparison': {
   474→      /* مدل ممکن است points یا items بنویسد — هر دو پذیرفته می‌شود */
   475→      const side = (v: unknown): { heading: string; points: string[] } | null => {
   476→        const o = (v ?? {}) as Record<string, unknown>
   477→        const heading = cut(o.heading, 40)
   478→        const rawPoints = Array.isArray(o.points) ? o.points : Array.isArray(o.items) ? o.items : []
   479→        const points = rawPoints.slice(0, 5).map((p) => cut(p, 110)).filter(Boolean)
   480→        if (!heading || points.length < 1) return null
   481→        return { heading, points }
   482→      }
   483→      const left = side(raw.left)
   484→      const right = side(raw.right)
   485→      if (!left || !right) return null
   486→      return {
   487→        type: 'comparison',
   488→        kicker: cut(raw.kicker, 40) || undefined,
   489→        title: cut(raw.title, 80),
   490→        left,
   491→        right,
   492→      }
   493→    }
   494→    default:
   495→      return null
   496→  }
   497→}

   499→export function normalizeDeck(
   500→  value: unknown,
   501→  language: 'fa' | 'en',
   502→  visualMode: 'offline' | 'image' = 'image',
   503→): DeckJSON | null {
   504→  if (!isValidDeck(value)) return null
   505→  const raw = value as DeckJSON
   506→  const slides = raw.slides
   507→    .map((s) => normSlide(s as unknown as Record<string, unknown>))
   508→    .filter((s): s is DeckSlide => s !== null)
   509→    .filter((s) => {
   510→      if (s.type === 'agenda') return s.items.length >= 2
   511→      if (s.type === 'bullets') return s.items.length >= 1
   512→      if (s.type === 'two-column') return s.columns.length === 2 && s.columns.every((c) => c.items.length >= 1)
   513→      if (s.type === 'stats') return s.stats.length >= 2
   514→      if (s.type === 'quote') return s.quote.length > 0
   515→      if (s.type === 'timeline') return s.steps.length >= 2
   516→      if (s.type === 'conclusion') return s.points.length >= 1
   517→      if (s.type === 'section') return s.title.trim().length > 0
   518→      if (s.type === 'comparison') return s.left.points.length >= 1 && s.right.points.length >= 1
   519→      return true
   520→    })
   521→    /* حالت آفلاین: هیچ تصویر خارجی نباید در فایل بماند */
   522→    .map((s) => {
   523→      if (visualMode !== 'offline') return s
   524→      if (s.type === 'image') return null
   525→      if (s.type === 'cover') return { ...s, image: undefined, imageCaption: undefined }
   526→      if (s.type === 'bullets')
   527→        return { ...s, items: s.items.map((it) => ({ ...it, image: undefined })) }
   528→      return s
   529→    })
   530→    .filter((s): s is DeckSlide => s !== null)
   531→  if (slides.length < 2) return null
   532→  const cover = slides[0].type === 'cover' ? slides[0] : null
   533→  const title = cover && 'title' in cover ? cover.title : raw.title
   534→  return {
   535→    title: cut(title, 120) || (language === 'fa' ? 'ارائه بدون عنوان' : 'Untitled deck'),
   536→    subtitle: cut(raw.subtitle, 200) || undefined,
   537→    language: raw.language === 'en' ? 'en' : language,
   538→    slides,
   539→  }
   540→}
   541→
   542→/* اصلاح ساختاری خروجی مدل: cover اول، conclusion آخر، و برش هوشمند
   543→ * تعداد اضافه (به‌جای بریدن از انتها، از میان اسلایدها کم می‌کنیم
   544→ * تا cover و conclusion همیشه سالم بمانند) */
   545→export function fixStructure(deck: DeckJSON, want: number): DeckJSON {
   546→  let slides = [...deck.slides]
   547→  const ci = slides.findIndex((s) => s.type === 'cover')
   548→  if (ci > 0) slides = [slides[ci], ...slides.slice(0, ci), ...slides.slice(ci + 1)]
   549→  const ei = slides.findIndex((s) => s.type === 'conclusion')
   550→  if (ei >= 0 && ei !== slides.length - 1) {
   551→    const [c] = slides.splice(ei, 1)
   552→    slides.push(c)
   553→  }
   554→  if (slides.length > want && slides.length >= 3) {
   555→    const first = slides[0]
   556→    const last = slides[slides.length - 1]
   557→    const mid = slides.slice(1, -1)
   558→    slides = [first, ...mid.slice(0, Math.max(1, want - 2)), last]
   559→  }
   560→  return { ...deck, slides }
   561→}

   564→
   565→export async function generateDeckWithAI(opts: GenerateOptions): Promise<DeckJSON> {
   566→  /* چیدمان قطعی: تعداد و نوع هر اسلاید از قبل برنامه‌ریزی می‌شود تا
   567→   * «بیش از ۸ اسلاید» هم دقیقاً رعایت شود و ترکیب انواع متنوع بماند */
   568→  const plan = planSlideTypes(opts.slideCount, opts.visualMode, topicSeed(opts.topic), opts.preferredLayouts)
   569→  const want = plan.length
   570→
   571→  let best: DeckJSON | null = null
   572→  let lastErr: unknown
   573→  // two attempts: LLM output can occasionally break JSON structure or come up short
   574→  for (let attempt = 0; attempt < 2; attempt++) {
   575→    try {
   576→      const content = await chat([
   577→        { role: 'system', content: systemPrompt(opts.language, opts.visualMode) },
   578→        { role: 'user', content: userPrompt(opts, plan) },
   579→      ])
   580→
   581→      let deck = normalizeDeck(extractJSON(content), opts.language, opts.visualMode)
   582→      if (!deck) throw new Error('ساختار ارائه تولیدشده معتبر نبود. دوباره تلاش کنید.')
   583→      deck = fixStructure(deck, want)
   584→
   585→      // تعداد رعایت شد؟ (یک اسلاید کمتر قابل قبول است)
   586→      if (deck.slides.length >= want - 1) return deck
   587→      // کمتر از درخواست — بهترین تلاش را نگه دار و یک‌بار دیگر بخواه
   588→      if (!best || deck.slides.length > best.slides.length) best = deck
   589→    } catch (err) {
   590→      lastErr = err
   591→    }
   592→  }
   593→  if (best) return best
   594→  throw lastErr instanceof Error
   595→    ? lastErr
   596→    : new Error('تولید پرزنتیشن ناموفق بود. لطفاً دوباره تلاش کنید.')
   597→}
