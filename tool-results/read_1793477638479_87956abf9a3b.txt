   152→function accentCss(accent: DeckAccent, light: boolean): string {

   154→  return `:root${light ? '.light' : ''}{--a-primary:${p.primary};--a-deep:${p.deep};--a-tint:${p.tint};--a-tintB:${p.tintB};--a-glow:${p.glow};--a-grad:${p.grad}}`
   155→}

   159→function head(slide: { kicker?: string; title: string }, deck: DeckJSON): string {

   166→function foot(deck: DeckJSON, idx: number, total: number): string {
   167→  const lang = deck.language
   168→  const n = lang === 'fa' ? faNum(idx + 1) : String(idx + 1)
   169→  const t = lang === 'fa' ? faNum(total) : String(total)
   170→  return `<footer class="foot"><span>${esc(deck.title)}</span><span class="n">${n} / ${t}</span></footer>`
   171→}

   438→function renderSlide(s: DeckSlide, deck: DeckJSON, idx: number): string {
   439→  switch (s.type) {
   440→    case 'cover': return renderCover(s, deck)
   441→    case 'agenda': return renderAgenda(s, deck, idx)
   442→    case 'bullets': return renderBullets(s, deck, idx)
   443→    case 'two-column': return renderTwoColumn(s, deck, idx)
   444→    case 'stats': return renderStats(s, deck, idx)
   445→    case 'quote': return renderQuote(s, deck, idx)
   446→    case 'image': return renderImage(s, deck, idx)
   447→    case 'timeline': return renderTimeline(s, deck, idx)
   448→    case 'conclusion': return renderConclusion(s, deck, idx)
   449→    case 'section': return renderSection(s, deck, idx)
   450→    case 'comparison': return renderComparison(s, deck, idx)
   451→    default: return renderBullets({ type: 'bullets', title: String((s as { title?: string }).title ?? deck.title), items: [] }, deck, idx)
   452→  }
   453→}
