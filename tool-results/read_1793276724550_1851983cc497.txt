   136→function accentCss(accent: DeckAccent, light: boolean): string {
   137→  const p = ACCENTS[accent][light ? 'light' : 'dark']
   138→  return `:root${light ? '.light' : ''}{--a-primary:${p.primary};--a-deep:${p.deep};--a-tint:${p.tint};--a-tintB:${p.tintB};--a-glow:${p.glow};--a-grad:${p.grad}}`

   143→function head(slide: { kicker?: string; title: string }, deck: DeckJSON): string {
   144→  const kicker = slide.kicker
   145→    ? `<div class="kicker"><span class="dot"></span>${esc(slide.kicker)}</div>`

   147→  return `<header class="head">${kicker}<h1 class="title">${esc(slide.title)}</h1><div class="rule"></div></header>`

   150→function foot(deck: DeckJSON, idx: number, total: number): string {
   151→  const lang = deck.language
   152→  const n = lang === 'fa' ? faNum(idx + 1) : String(idx + 1)
   153→  const t = lang === 'fa' ? faNum(total) : String(total)
   154→  return `<footer class="foot"><span>${esc(deck.title)}</span><span class="n">${n} / ${t}</span></footer>`
   155→}

   285→  <div class="body"><section style="flex:1;display:flex;gap:16px;align-items:stretch;padding-top:8px">${steps}</section></div>

   289→function renderConclusion(s: Extract<DeckSlide, { type: 'conclusion' }>, deck: DeckJSON, idx: number): string {

   306→function renderSlide(s: DeckSlide, deck: DeckJSON, idx: number): string {
   307→  switch (s.type) {
   308→    case 'cover': return renderCover(s, deck)
   309→    case 'agenda': return renderAgenda(s, deck, idx)
   310→    case 'bullets': return renderBullets(s, deck, idx)
   311→    case 'two-column': return renderTwoColumn(s, deck, idx)
   312→    case 'stats': return renderStats(s, deck, idx)
   313→    case 'quote': return renderQuote(s, deck, idx)
   314→    case 'timeline': return renderTimeline(s, deck, idx)
   315→    case 'conclusion': return renderConclusion(s, deck, idx)
   316→    default: return renderBullets({ type: 'bullets', title: String((s as { title?: string }).title ?? deck.title), items: [] }, deck, idx)
   317→  }
   318→}
