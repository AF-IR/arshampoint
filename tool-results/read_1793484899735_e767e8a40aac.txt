   177→function head(slide: { kicker?: string; title: string }, deck: DeckJSON): string {
   178→  const kicker = slide.kicker
   179→    ? `<div class="kicker"><span class="dot"></span>${esc(slide.kicker)}</div>`
   180→    : ''
   181→  return `<header class="head">${kicker}<h1 class="title">${esc(slide.title)}</h1><div class="rule"></div></header>`
   182→}

   184→function foot(deck: DeckJSON, idx: number, total: number): string {
   185→  const lang = deck.language
   186→  const n = lang === 'fa' ? faNum(idx + 1) : String(idx + 1)
   187→  const t = lang === 'fa' ? faNum(total) : String(total)
   188→  return `<footer class="foot"><span>${esc(deck.title)}</span><span class="n">${n} / ${t}</span></footer>`
   189→}

   456→function renderSlide(s: DeckSlide, deck: DeckJSON, idx: number): string {
   457→  switch (s.type) {
   458→    case 'cover': return renderCover(s, deck)
   459→    case 'agenda': return renderAgenda(s, deck, idx)
   460→    case 'bullets': return renderBullets(s, deck, idx)
   461→    case 'two-column': return renderTwoColumn(s, deck, idx)
   462→    case 'stats': return renderStats(s, deck, idx)
   463→    case 'quote': return renderQuote(s, deck, idx)
   464→    case 'image': return renderImage(s, deck, idx)
   465→    case 'timeline': return renderTimeline(s, deck, idx)
   466→    case 'conclusion': return renderConclusion(s, deck, idx)
   467→    case 'section': return renderSection(s, deck, idx)
   468→    case 'comparison': return renderComparison(s, deck, idx)
   469→    default: return renderBullets({ type: 'bullets', title: String((s as { title?: string }).title ?? deck.title), items: [] }, deck, idx)
   470→  }
   471→}
