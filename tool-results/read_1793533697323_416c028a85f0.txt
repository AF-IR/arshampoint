   551→   551→function renderSlide(s: DeckSlide, deck: DeckJSON, idx: number, styleKey: DeckStyle): string {
   552→   552→  switch (s.type) {
   553→   553→    case 'cover': return renderCover(s, deck, styleKey)
   554→   554→    case 'agenda': return renderAgenda(s, deck, idx)
   555→   555→    case 'bullets': return renderBullets(s, deck, idx)
   556→   556→    case 'two-column': return renderTwoColumn(s, deck, idx)
   557→   557→    case 'stats': return renderStats(s, deck, idx)
   558→   558→    case 'quote': return renderQuote(s, deck, idx)
   559→   559→    case 'image': return renderImage(s, deck, idx)
   560→   560→    case 'timeline': return renderTimeline(s, deck, idx)
   561→   561→    case 'conclusion': return renderConclusion(s, deck, idx)
   562→   562→    case 'section': return renderSection(s, deck, idx, styleKey)
   563→   563→    case 'comparison': return renderComparison(s, deck, idx)
   564→   564→    default: return renderBullets({ type: 'bullets', title: String((s as { title?: string }).title ?? deck.title), items: [] }, deck, idx)
   565→   565→  }
