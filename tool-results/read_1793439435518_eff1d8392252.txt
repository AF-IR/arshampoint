   143→function head(slide: { kicker?: string; title: string }, deck: DeckJSON): string {

   150→function foot(deck: DeckJSON, idx: number, total: number): string {
   151→  const lang = deck.language
   152→  const n = lang === 'fa' ? faNum(idx + 1) : String(idx + 1)
   153→  const t = lang === 'fa' ? faNum(total) : String(total)
   154→  return `<footer class="foot"><span>${esc(deck.title)}</span><span class="n">${n} / ${t}</span></footer>`
   155→}

   158→function imgTag(url: string, alt: string, seed: string, style = ''): string {
   159→  const fallback = `https://picsum.photos/seed/${encodeURIComponent(seed)}/800/500`
   160→  const safeAlt = esc(alt).slice(0, 120)
   161→  return `<img src="${esc(url)}" alt="${safeAlt}" loading="lazy" referrerpolicy="no-referrer" onerror="this.onerror=null;this.src='${fallback}'" style="object-fit:cover;width:100%;height:100%;display:block;${style}">`
   162→}

   197→function renderAgenda(s: Extract<DeckSlide, { type: 'agenda' }>, deck: DeckJSON, idx: number): string {
   198→  const lang = deck.language
   199→  const two = s.items.length > 4
   200→  const items = s.items
   201→    .map((it, i) => {
   202→      const n = lang === 'fa' ? faNum(i + 1) : String(i + 1)
   203→      return `<div class="card" style="display:flex;align-items:center;gap:18px;padding:20px 24px">
   204→        <span class="num" style="width:44px;height:44px;border-radius:14px;background:var(--a-tint);border:1px solid var(--a-tintB);color:var(--a-primary);font-weight:800;font-size:19px;display:grid;place-items:center;flex-shrink:0">${n}</span>
   205→        <span style="font-size:17.5px;font-weight:600;line-height:1.7">${esc(it)}</span>
   206→      </div>`
   207→    })
   208→    .join('')
   209→  return `${head(s, deck)}
   210→  <div class="body"><section style="flex:1;display:grid;grid-template-columns:${two ? '1fr 1fr' : '1fr'};gap:16px;align-content:${two ? 'stretch' : 'start'}">${items}</section></div>
   211→  ${foot(deck, idx, deck.slides.length)}`
   212→}

   345→function renderSlide(s: DeckSlide, deck: DeckJSON, idx: number): string {
   346→  switch (s.type) {
   347→    case 'cover': return renderCover(s, deck)
   348→    case 'agenda': return renderAgenda(s, deck, idx)
   349→    case 'bullets': return renderBullets(s, deck, idx)
   350→    case 'two-column': return renderTwoColumn(s, deck, idx)
   351→    case 'stats': return renderStats(s, deck, idx)
   352→    case 'quote': return renderQuote(s, deck, idx)
   353→    case 'image': return renderImage(s, deck, idx)
   354→    case 'timeline': return renderTimeline(s, deck, idx)
   355→    case 'conclusion': return renderConclusion(s, deck, idx)
   356→    default: return renderBullets({ type: 'bullets', title: String((s as { title?: string }).title ?? deck.title), items: [] }, deck, idx)
   357→  }
   358→}

   442→  },{passive:true});
   443→  try{if(localStorage.getItem('zslide-theme')==='light')document.documentElement.classList.add('light')}catch(e){}
   444→  go(0);
