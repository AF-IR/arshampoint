   185→function foot(deck: DeckJSON, idx: number, total: number): string {
   186→  const lang = deck.language
   187→  const n = lang === 'fa' ? faNum(idx + 1) : String(idx + 1)
   188→  const t = lang === 'fa' ? faNum(total) : String(total)
   189→  return `<footer class="foot"><span>${esc(deck.title)}</span><span class="n">${n} / ${t}</span></footer>`

   571→function renderSlide(s: DeckSlide, deck: DeckJSON, idx: number, styleKey: DeckStyle): string {
   572→  switch (s.type) {
   573→    case 'cover': return renderCover(s, deck, styleKey)
   574→    case 'agenda': return renderAgenda(s, deck, idx)
   575→    case 'bullets': return renderBullets(s, deck, idx)
   576→    case 'two-column': return renderTwoColumn(s, deck, idx)
   577→    case 'stats': return renderStats(s, deck, idx)
   578→    case 'quote': return renderQuote(s, deck, idx)
   579→    case 'image': return renderImage(s, deck, idx)
   580→    case 'timeline': return renderTimeline(s, deck, idx)
   581→    case 'conclusion': return renderConclusion(s, deck, idx)
   582→    case 'section': return renderSection(s, deck, idx, styleKey)
   583→    case 'comparison': return renderComparison(s, deck, idx)
   584→    default: return renderBullets({ type: 'bullets', title: String((s as { title?: string }).title ?? deck.title), items: [] }, deck, idx)
   585→  }
   586→}
